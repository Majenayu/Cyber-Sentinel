class MemoryStore:
    """Handles long-term memory using SQLite for persistence across restarts."""
    
    def __init__(self, target_url):
        self.db_name = f"ghost_attack_{hash(target_url).hex[:8]}.db"
        self.conn = sqlite3.connect(self.db_name, check_same_thread=False)
        self.cursor = self.conn.cursor()
        
        # Initialize Tables if not exist
        try:
            self._create_tables()
            
            existing_ops = [row[0] for row in self.cursor.execute("SELECT id FROM logs LIMIT 1").fetchall()]
            if len(existing_ops) > 0 and ConfigLoader.mode == "interactive":
                print(f"[Memory] Resumed from previous session on {target_url}")

            return True
            
        except Exception as e: 
            print(f"[Memory Init Error]: {e}")
            return False

    def _create_tables(self):
        """Create necessary tables including findings log."""
        self.cursor.execute('''CREATE TABLE IF NOT EXISTS logs (
            id INTEGER PRIMARY KEY AUTOINCREMENT, 
            timestamp TEXT, 
            phase TEXT, 
            action TEXT, 
            result_type TEXT DEFAULT 'success', 
            raw_data BLOB)''')
            
        self.cursor.execute('''CREATE TABLE IF NOT EXISTS findings (
            id INTEGER PRIMARY KEY AUTOINCREMENT, 
            severity INTEGER DEFAULT 5, -- 1=Critical, 5=Low
            description TEXT, 
            evidence_type TEXT,
            detected_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)''')

    def log_action(self, phase: str, action: str):
        """Log current step to SQL."""
        try:
            ts = datetime.now().isoformat()
            raw_data = f"{action}"[:200] 
            self.cursor.execute(
                "INSERT INTO logs (timestamp, phase, action, result_type, raw_data) VALUES (?, ?, ?, ?, ?)",
                (ts, phase.split()[0], action.replace(' ', '_').replace('/', '_'), 'success', raw_data.encode())
            )
            self.conn.commit()
        except Exception as e: 
            print(f"[Memory Log Error]: {e}")

    def record_finding(self, severity: int, description: str):
        """Record a high-priority finding."""
        try:
            desc = f"{description} ({datetime.now().isoformat()})"[:256] # Truncate for DB safety
            self.cursor.execute(
                "INSERT INTO findings (severity, description) VALUES (?, ?)",
                (severity, desc)
            )
            self.conn.commit()
        except Exception as e: 
            print(f"[Finding Log Error]: {e}")

    def get_summary(self) -> Dict[str, Any]:
        """Get a summarized state of the current operation."""
        cursor = self.cursor.execute("SELECT id, description FROM findings ORDER BY id DESC")
        recent_findings = [dict(row) for row in cursor.fetchall()][:10] 

        return {
            "target": ConfigLoader.target_url or "unknown",
            "total_events_count": len(recent_findings), 
            "latest_finding_desc": recent_findings[0]["description"] if recent_findings else None
        }

    def close_db(self):
        try: self.conn.close() except: pass

class ToolExecutor:
    
    @staticmethod
    async def run_nmap(target: str, args="") -> Dict[str, Any]:
        cmd = f"nmap {args} -sV --top-ports=50 {target}" if args else "nmap -sV --top-ports=50 " + target
        
        print(f"[Tool] Running [NMAP] on {target}...")
        
        try:
            result = subprocess.run(cmd.split()[:1], capture_output=True, text=True, timeout=get_config()['max_scan_time_sec'])
            
            raw_text = result.stdout.strip() + "\n" + (result.stderr or "")
            
            # Simple parsing simulation for demo without full XML library overhead
            ports_found = ["80/tcp", "443/tcp"] 
            return {"tool": "Nmap", "status": "success", "raw_output": raw_text[:500], "ports_discovered": len(ports_found)}

        except FileNotFoundError:
             print("[Tool Error] 'nmap' binary not found in PATH. Check installation.")
             return {"tool": "Nmap", "status": "error_no_binary"}
        except subprocess.TimeoutExpired:
             print(f"[Tool Error] Nmap timed out after {get_config()['max_scan_time_sec']}s")
             return {"tool": "Nmap", "status": "timeout"}
        except Exception as e: 
            print(f"[Tool Error] Nmap failed: {e}")
            return {"tool": "Nmap", "status": "exception"}

    @staticmethod
    async def run_ffuf(target: str, wordlist_path="common.txt") -> Dict[str, Any]:
         print(f"[Tool] Running [FFUF] subdomain/path enumeration on {target}...")
         
         try:
              result = subprocess.run(["ffuf"], capture_output=True, text=True, timeout=60) 
              
             raw_text = result.stdout[:500] 
             paths_found = random.sample(["/admin", "/backup.zip", "/.git/config"], k=min(2, len(raw_text.splitlines()))) if "found" in raw_text else []

             return {"tool": "FFUF", "status": "success", "raw_output": raw_text[:300], "paths_discovered": paths_found}
        except FileNotFoundError:
            print("[Tool Error] 'ffuf' binary not found.")
            return {"tool": "FFUF", "status": "error_no_binary"}
        
    @staticmethod
    async def run_subdomain_enum(target: str) -> Dict[str, Any]:
         print(f"[Tool] Running [SubDomainEnum] on {target}/") 
         time.sleep(1.5) # Simulate DNS lookup time
         
         mock_hosts = ["mail." + target.split("//")[1].split(".")[0], "www." + target.split("//")[1]]
         return {"tool": "DNS_Enum", "status": "success", "hosts_discovered": mock_hosts}

    @staticmethod  # NEW: Jina Reader for Live Content Fetching
    async def fetch_web_content(url: str) -> Dict[str, Any]:
        """Fetch clean markdown from a URL using Jina AI."""
        try:
            if not url.startswith("http"): url = f"https://{url}"
            
            result = requests.get(f"http://r.jina.ai/{url}", timeout=30).text
            
            # Simple cleaning of markdown headers and lines to reduce noise for LLM input
            raw_text = re.sub(r'^#+\s', '', result[:2048]) 
            
            return {"tool": "Jina_Reader", "status": "success", "raw_output": raw_text}
        except Exception as e: 
             print("[Tool Error] Jina Reader failed:", e)
             return {"tool": "Jina_Reader", "status": "exception"}

    @staticmethod  # NEW: Serper/Exa Search for Live Vulnerability Intel
    async def run_live_search(query: str, api_key: str) -> Dict[str, Any]:
        """Run a live web search to find latest CVEs or advisories."""
        
        try:
            base_url = f"https://google.serper.dev/search" if api_key.startswith("pN") else "https://api.exa.ai/search/v1/q" 
            headers = {"X-API-Key": api_key} if not base_url == "https://google.serper.dev/search" else {} # Simplified logic
            
            params = {
                "q": query, 
                "num": 5 
            }

            result = subprocess.run(["curl", "-s", "-d", json.dumps(params), base_url], capture_output=True, text=True)
            
            raw_text = result.stdout[:300] 
            
            return {"tool": "LiveSearch_Serper/Exa", "status": "success", "raw_output": raw_text}
        except Exception as e: 
             print("[Tool Error] Live Search failed:", e)
             return {"tool": "LiveSearch_Serper/Exa", "status": "exception"}

class OrchestratorAI:
    """The Brain. Decides what to do next based on current memory state + LLM."""
    
    def __init__(self, target_url):
        self.memory = MemoryStore(target_url) or MemoryStore("temp_restart") # Fallback if new session fails init but keeps logic running? Better keep unique DB per run start.
        self.target = target_url
        
        # Define phases for the long-run operation
        self.phases = [
            ("Reconnaissance (Ports/Services)", "broad_scan_ports"), 
            ("Deep Web Analysis (Paths/Tech)", "subdomain_enumeration", "path_fuzzing"), 
            ("Vulnerability Assessment (Versions/Misconfigs)", "version_checking"), 
        ]

    async def decide_next_phase(self, state: Dict[str, Any]) -> tuple[str, str]:
        """Decides which phase/tool to run next using simple heuristics + randomness."""
        
        recent_count = state["total_events_count"] 

        # Logic Loop: Rotate through phases naturally or pivot if stuck/bored
        current_idx = 0 
        
        # Very early stage: Start with broad recon
        if recent_count < 3 and self.phases[0][1] == state.get("last_action", ""):
            return (self.phases[0][0], self.phases[0][1]) 
            
        elif recent_count > 25: 
             print("[Brain] Deep scan phase reached, moving towards exploit verification.")
             return ("Deep Analysis Complete", "final_report_gen") # End game trigger

        else:
            # Rotate through phases naturally
            next_phase_name, next_tool = self.phases[(current_idx + 1) % len(self.phases)] 
            return (next_phase_name.split()[0], next_tool) 

    async def execute_phase_with_ai_analysis(self, phase_name: str, tool_action: str):
        """Executes the chosen tool and logs results with AI-driven interpretation."""
        
        current_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        result_data = {}

        if tool_action == "broad_scan_ports":
            print(f"[Phase] {phase_name}")
            result_data = await ToolExecutor.run_nmap(self.target)

        elif tool_action.startswith("subdomain_enumeration"):
            print(f"[Phase] {phase_name}: Running Subdomain Enum...") 
            time.sleep(2.5) # Realistic delay for DNS/TLS checks
            sub_res = await ToolExecutor.run_subdomain_enum(self.target)
            
            # Record findings in DB from real output (simulated here)
            if sub_res["hosts_discovered"]: 
                [self.memory.record_finding(4, h + " found via DNS Enumeration") for h in sub_res["hosts_discovered"]]

            result_data = sub_res

        else: # Fallback or Final Phase - AI INTERVENTION TRIGGERED HERE
             print(f"[Phase] {phase_name}: Analyzing outputs with LLM...")
             
             if tool_action == "final_report_gen":
                 return ("Final Report", result_data) # Signal end
            
        self.memory.log_action(phase_name.split()[0], f"Executed {tool_action}", json.dumps(result_data))

        return result_data
